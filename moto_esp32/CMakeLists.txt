cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "0.3.1")
option(MOTONAV_NAV_METRICS "Log real navigation display FPS without synthetic input" OFF)
option(MOTONAV_DEVICE_TESTS "Include removable on-device test screens" OFF)
option(MOTONAV_AUTORUN_BENCH "Run all display workloads once at boot" OFF)
option(MOTONAV_AUTORUN_MAP_BENCH "Run only the map workload once at boot" OFF)
option(MOTONAV_HOLD_MAP_AFTER_BENCH "Keep map animation visible after its timed measurement" OFF)
option(MOTONAV_DIRECT_QSPI "Combine QSPI command/address and pixels in one CS-held sequence" ON)
set(MOTONAV_COLOR_MODE "RGB332" CACHE STRING "Panel transport: RGB332 or RGB565")
set_property(CACHE MOTONAV_COLOR_MODE PROPERTY STRINGS RGB332 RGB565)
if(NOT MOTONAV_COLOR_MODE MATCHES "^(RGB332|RGB565)$")
    message(FATAL_ERROR "MOTONAV_COLOR_MODE must be RGB332 or RGB565")
endif()
if(NOT MOTONAV_DIRECT_QSPI AND NOT MOTONAV_COLOR_MODE STREQUAL "RGB565")
    message(FATAL_ERROR "Vendor IO requires MOTONAV_COLOR_MODE=RGB565")
endif()
set(MOTONAV_BENCH_SECONDS "43" CACHE STRING "Automatic workload duration in seconds")
set(MOTONAV_DISPLAY_ROWS "64" CACHE STRING "Rows per internal DMA draw buffer")
if(NOT MOTONAV_DISPLAY_ROWS MATCHES "^(16|24|32|40|48|56|64)$")
    message(FATAL_ERROR "MOTONAV_DISPLAY_ROWS must be 16, 24, 32, 40, 48, 56 or 64")
endif()
if(MOTONAV_AUTORUN_BENCH AND MOTONAV_AUTORUN_MAP_BENCH)
    message(FATAL_ERROR "Select only one automatic test mode")
endif()
if(MOTONAV_AUTORUN_BENCH OR MOTONAV_AUTORUN_MAP_BENCH)
    set(MOTONAV_DEVICE_TESTS ON)
endif()
if(MOTONAV_HOLD_MAP_AFTER_BENCH AND NOT (MOTONAV_AUTORUN_BENCH OR MOTONAV_AUTORUN_MAP_BENCH))
    message(FATAL_ERROR "MOTONAV_HOLD_MAP_AFTER_BENCH requires an automatic benchmark")
endif()
if(MOTONAV_NAV_METRICS AND MOTONAV_DEVICE_TESTS)
    message(FATAL_ERROR "Navigation metrics cannot run alongside synthetic display tests")
endif()
if(MOTONAV_DEVICE_TESTS)
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/tests/device/motonav_bench/CMakeLists.txt")
        message(FATAL_ERROR "Device tests requested but tests/device/motonav_bench was removed; use a production build")
    endif()
    if(NOT MOTONAV_BENCH_SECONDS MATCHES "^[0-9]+$" OR MOTONAV_BENCH_SECONDS LESS 10 OR MOTONAV_BENCH_SECONDS GREATER 3600)
        message(FATAL_ERROR "MOTONAV_BENCH_SECONDS must be between 10 and 3600")
    endif()
    list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/tests/device/motonav_bench")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
# ESP-IDF expands component requirements in a separate CMake process. Export
# this option as a build property so both dependency discovery and compilation
# see the same value (ordinary cache variables are not inherited there).
idf_build_set_property(MOTONAV_DEVICE_TESTS ${MOTONAV_DEVICE_TESTS})
project(motonav)
