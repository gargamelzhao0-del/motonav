option(MOTONAV_AUTORUN_MAP_BENCH "Run only the map workload once at boot (bench firmware only)" OFF)
set(MOTONAV_BENCH_SECONDS "43" CACHE STRING "Automatic workload duration in seconds")
if(NOT MOTONAV_BENCH_SECONDS MATCHES "^[0-9]+$" OR MOTONAV_BENCH_SECONDS LESS 10 OR MOTONAV_BENCH_SECONDS GREATER 3600)

endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE MOTONAV_BENCH_SECONDS=${MOTONAV_BENCH_SECONDS})
if(MOTONAV_AUTORUN_BENCH)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE MOTONAV_AUTORUN_BENCH=1)
endif()
if(MOTONAV_AUTORUN_MAP_BENCH)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE MOTONAV_AUTORUN_MAP_BENCH=1)
endif()
