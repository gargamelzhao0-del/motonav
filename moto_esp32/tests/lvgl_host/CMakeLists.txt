cmake_minimum_required(VERSION 3.16)
project(motonav_ui_preview LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
set(COMPONENTS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../components")
set(BENCH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../device/motonav_bench")
function(add_preview target tests)
    add_executable(${target} preview.cpp
        ${COMPONENTS_DIR}/motonav_ui/navigation_ui.cpp ${COMPONENTS_DIR}/motonav_ui/dashboard.cpp
        ${COMPONENTS_DIR}/motonav_ui/label_cache.cpp)
    target_include_directories(${target} PRIVATE .
        ${COMPONENTS_DIR}/motonav_core/include ${COMPONENTS_DIR}/motonav_ui/include)
    target_compile_definitions(${target} PRIVATE MOTONAV_DEVICE_TESTS=${tests})
    if(tests)
        target_sources(${target} PRIVATE ${COMPONENTS_DIR}/motonav_graphics/route_map.cpp
            ${BENCH_DIR}/display_benchmark.cpp ${BENCH_DIR}/map_fixture.cpp)
        target_include_directories(${target} PRIVATE
            ${COMPONENTS_DIR}/motonav_graphics/include ${BENCH_DIR}/include)
    endif()
    target_link_libraries(${target} PRIVATE lvgl)
endfunction()
add_preview(ui_production 0)
add_preview(ui_preview 1)
add_preview(ui_autorun_test 1)
target_compile_definitions(ui_autorun_test PRIVATE MOTONAV_AUTORUN_BENCH=1)
add_preview(ui_map_autorun_test 1)
target_compile_definitions(ui_map_autorun_test PRIVATE MOTONAV_AUTORUN_MAP_BENCH=1)
add_preview(ui_map_hold_test 1)
target_compile_definitions(ui_map_hold_test PRIVATE MOTONAV_AUTORUN_MAP_BENCH=1 MOTONAV_HOLD_MAP_AFTER_BENCH=1)
add_preview(ui_map_decoder_test 1)
target_compile_definitions(ui_map_decoder_test PRIVATE MOTONAV_MAP_DECODER_ONLY=1)
add_preview(ui_uncached_test 1)
target_compile_definitions(ui_uncached_test PRIVATE MOTONAV_DISABLE_LABEL_CACHE=1)
add_preview(ui_rgb332_preview 1)
add_preview(ui_rgb332_reference 1)
target_compile_definitions(ui_rgb332_preview PRIVATE MOTONAV_PREVIEW_RGB332=1)
target_compile_definitions(ui_rgb332_reference PRIVATE MOTONAV_PREVIEW_RGB332=2)
foreach(target ui_rgb332_preview ui_rgb332_reference)
    target_include_directories(${target} PRIVATE ${COMPONENTS_DIR}/motonav_display)
    target_sources(${target} PRIVATE ${COMPONENTS_DIR}/motonav_display/pixel_pack.cpp)
endforeach()
enable_testing()
add_test(NAME ui_production_without_tests COMMAND ui_production "${CMAKE_CURRENT_BINARY_DIR}/preview-production")
add_test(NAME ui_normal COMMAND ui_preview "${CMAKE_CURRENT_BINARY_DIR}/preview-normal")
add_test(NAME ui_all_workloads COMMAND ui_autorun_test "${CMAKE_CURRENT_BINARY_DIR}/preview-all")
add_test(NAME ui_map_workload COMMAND ui_map_autorun_test "${CMAKE_CURRENT_BINARY_DIR}/preview-map")
add_test(NAME ui_map_remains_after_measurement COMMAND ui_map_hold_test "${CMAKE_CURRENT_BINARY_DIR}/preview-map-hold")
add_test(NAME ui_decoder_fallback COMMAND ui_map_decoder_test "${CMAKE_CURRENT_BINARY_DIR}/preview-decoder")
add_test(NAME ui_map_pixels_equal COMMAND ${CMAKE_COMMAND} -E compare_files
    "${CMAKE_CURRENT_BINARY_DIR}/preview-normal/map-moving.ppm"
    "${CMAKE_CURRENT_BINARY_DIR}/preview-decoder/map-moving.ppm")
set_tests_properties(ui_map_pixels_equal PROPERTIES DEPENDS "ui_normal;ui_decoder_fallback")
add_test(NAME ui_uncached_reference COMMAND ui_uncached_test "${CMAKE_CURRENT_BINARY_DIR}/preview-uncached")
add_test(NAME ui_cached_pixels_equal COMMAND ${CMAKE_COMMAND}
    "-DROOT=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/compare_cache.cmake")
set_tests_properties(ui_cached_pixels_equal PROPERTIES DEPENDS "ui_normal;ui_uncached_reference")
add_test(NAME ui_rgb332_preview COMMAND ui_rgb332_preview "${CMAKE_CURRENT_BINARY_DIR}/preview-rgb332")
add_test(NAME ui_rgb332_reference COMMAND ui_rgb332_reference "${CMAKE_CURRENT_BINARY_DIR}/preview-rgb332-reference")
add_test(NAME ui_rgb332_pixels_equal COMMAND ${CMAKE_COMMAND}
    "-DROOT=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/compare_rgb332.cmake")
set_tests_properties(ui_rgb332_pixels_equal PROPERTIES DEPENDS "ui_rgb332_preview;ui_rgb332_reference")
