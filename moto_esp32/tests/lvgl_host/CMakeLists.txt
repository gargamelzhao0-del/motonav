cmake_minimum_required(VERSION 3.16)
project(motonav_ui_preview LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(ui_preview preview.cpp ../../main/navigation_ui.cpp)
target_include_directories(ui_preview PRIVATE . ../../main)
target_link_libraries(ui_preview PRIVATE lvgl)
add_executable(ui_autorun_test preview.cpp ../../main/navigation_ui.cpp)
target_include_directories(ui_autorun_test PRIVATE . ../../main)
target_compile_definitions(ui_autorun_test PRIVATE MOTONAV_AUTORUN_BENCH=1)
target_link_libraries(ui_autorun_test PRIVATE lvgl)
